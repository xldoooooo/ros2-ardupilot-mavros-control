# Build the independent bridge and its pure map regression test.
cmake_minimum_required(VERSION 3.16)
project(avoidance_bridge)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp sensor_msgs nav_msgs geometry_msgs guided_interfaces correction_interfaces path_planning tf2)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(avoidance_bridge_node src/avoidance_bridge.cpp)
target_include_directories(avoidance_bridge_node PRIVATE include)
ament_target_dependencies(avoidance_bridge_node rclcpp sensor_msgs nav_msgs geometry_msgs guided_interfaces correction_interfaces path_planning tf2)
install(TARGETS avoidance_bridge_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_observed_map test/test_observed_map.cpp)
  target_include_directories(test_observed_map PRIVATE include)
  target_compile_options(test_observed_map PRIVATE -UNDEBUG)
  add_test(NAME observed_map COMMAND test_observed_map)
  add_executable(test_input_contract test/test_input_contract.cpp)
  target_include_directories(test_input_contract PRIVATE include)
  target_compile_options(test_input_contract PRIVATE -UNDEBUG)
  ament_target_dependencies(test_input_contract sensor_msgs)
  add_test(NAME input_contract COMMAND test_input_contract)
endif()
ament_package()
