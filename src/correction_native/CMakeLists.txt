# Build the native data plane against the same OpenCV version as Python.
cmake_minimum_required(VERSION 3.19)
project(correction_native)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11_vendor REQUIRED)
find_package(pybind11 REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_opencv.py"
  --cache "${CMAKE_CURRENT_BINARY_DIR}/opencv" --output "${CMAKE_CURRENT_BINARY_DIR}/opencv.cmake"
  COMMAND_ERROR_IS_FATAL ANY)
include("${CMAKE_CURRENT_BINARY_DIR}/opencv.cmake")
message(STATUS "Correction OpenCV ${CORRECTION_OPENCV_VERSION}: ${CORRECTION_OPENCV_INCLUDE}")
pybind11_add_module(_correction_native NO_EXTRAS src/bindings.cpp src/odometry.cpp src/vision.cpp)
target_include_directories(_correction_native PRIVATE include ${CORRECTION_OPENCV_INCLUDE})
ament_target_dependencies(_correction_native PUBLIC rclcpp nav_msgs)
target_link_libraries(_correction_native PRIVATE ${CORRECTION_OPENCV_LIBRARIES})
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
install(TARGETS _correction_native LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}")
add_executable(uvc_camera_node src/uvc_camera.cpp)
target_include_directories(uvc_camera_node PRIVATE include ${CORRECTION_OPENCV_INCLUDE})
ament_target_dependencies(uvc_camera_node rclcpp sensor_msgs)
target_link_libraries(uvc_camera_node ${CORRECTION_OPENCV_LIBRARIES})
install(TARGETS uvc_camera_node DESTINATION lib/${PROJECT_NAME})
ament_package()
